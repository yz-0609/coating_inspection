cmake_minimum_required(VERSION 3.21)
# Standalone configure: do not inherit the firmware's Arm toolchain.
project(coating_mcu_host_tests LANGUAGES C CXX)
if(CMAKE_CROSSCOMPILING OR CMAKE_C_COMPILER MATCHES "arm-none-eabi")
    message(FATAL_ERROR "Host tests require native GCC and a separate build directory")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
include(../cmake/pure_modules.cmake)
add_library(mcu_logic STATIC ${PURE_MODULE_SOURCES})
target_include_directories(mcu_logic PUBLIC ${PURE_MODULE_INCLUDES})
target_compile_options(mcu_logic PRIVATE -Wall -Wextra -Werror -pedantic -fshort-enums)
foreach(name reset_reason modbus workstation inputs_health transport pressure)
    add_executable(test_${name} test_${name}.c)
    target_link_libraries(test_${name} PRIVATE mcu_logic)
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME ${name} COMMAND test_${name})
endforeach()
add_executable(test_host_contract test_host_contract.cpp vendor/host_protocol/src/workstation/protocol.cpp)
target_link_libraries(test_host_contract PRIVATE mcu_logic)
target_include_directories(test_host_contract PRIVATE vendor/host_protocol/include)
target_compile_features(test_host_contract PRIVATE cxx_std_17)
target_compile_options(test_host_contract PRIVATE -Wall -Wextra -Werror -pedantic)
add_test(NAME upstream_host_contract COMMAND test_host_contract)
add_executable(mcu_emulator mcu_emulator.c)
target_link_libraries(mcu_emulator PRIVATE mcu_logic)
target_compile_options(mcu_emulator PRIVATE -Wall -Wextra -Werror -pedantic)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME offline_replay COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/../tools/offline_replay.py --emulator $<TARGET_FILE:mcu_emulator>)
set_tests_properties(pressure offline_replay PROPERTIES TIMEOUT 30)
